=== Exact default leaderboard SQL ===
SELECT leaderboard.participant_id, leaderboard.user_id, leaderboard.solved, leaderboard.penalty, leaderboard.last_solved_at, leaderboard.last_null 
FROM (SELECT contest_participants.id AS participant_id, contest_participants.user_id AS user_id, count(anon_1.sid) AS solved, CAST(coalesce(sum(timestampdiff(MINUTE, '2026-11-18 10:15:30.537942', anon_1.sat) + coalesce((SELECT count(submissions.id) AS count_1 
FROM submissions INNER JOIN judge_attempts ON judge_attempts.id = submissions.authoritative_attempt_id INNER JOIN verdicts ON verdicts.judge_attempt_id = judge_attempts.id 
WHERE submissions.contest_participant_id = contest_participants.id AND submissions.contest_problem_id = anon_1.pid AND verdicts.verdict != 'ACCEPTED' AND verdicts.verdict != 'JUDGE_INFRA_FAILURE' AND (submissions.submitted_at < anon_1.sat OR submissions.submitted_at = anon_1.sat AND submissions.id < anon_1.sid)), 0) * 20), 0) AS SIGNED INTEGER) AS penalty, max(anon_1.sat) AS last_solved_at, CASE WHEN (max(anon_1.sat) IS NULL) THEN 1 ELSE 0 END AS last_null 
FROM contest_participants LEFT OUTER JOIN (SELECT anon_2.uid AS uid, anon_2.pid AS pid, anon_2.sid AS sid, anon_2.sat AS sat 
FROM (SELECT contest_participants.user_id AS uid, submissions.contest_problem_id AS pid, submissions.id AS sid, submissions.submitted_at AS sat, row_number() OVER (PARTITION BY contest_participants.user_id, submissions.contest_problem_id ORDER BY submissions.submitted_at, submissions.id) AS rn 
FROM contest_participants INNER JOIN submissions ON submissions.contest_participant_id = contest_participants.id INNER JOIN judge_attempts ON judge_attempts.id = submissions.authoritative_attempt_id INNER JOIN verdicts ON verdicts.judge_attempt_id = judge_attempts.id 
WHERE contest_participants.contest_id = 403 AND contest_participants.status = 'registered' AND verdicts.verdict = 'ACCEPTED') AS anon_2 
WHERE anon_2.rn = 1) AS anon_1 ON anon_1.uid = contest_participants.user_id 
WHERE contest_participants.contest_id = 403 AND contest_participants.status = 'registered' GROUP BY contest_participants.id, contest_participants.user_id) AS leaderboard ORDER BY leaderboard.solved DESC, leaderboard.penalty ASC, leaderboard.last_null ASC, leaderboard.last_solved_at ASC, leaderboard.participant_id ASC 
 LIMIT 51
